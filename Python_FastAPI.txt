## 36. Complete CRUD API

```python
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel

app = FastAPI()


class Product(BaseModel):
    name: str
    price: float


products = {}


@app.post("/products")
def create_product(product_id: int, product: Product):

    if product_id in products:
        raise HTTPException(409, "Product already exists")

    products[product_id] = product.model_dump()

    return {"message": "Product created"}


@app.get("/products")
def get_products():
    return products


@app.get("/products/{product_id}")
def get_product(product_id: int):

    if product_id not in products:
        raise HTTPException(404, "Product not found")

    return products[product_id]


@app.put("/products/{product_id}")
def update_product(product_id: int, product: Product):

    if product_id not in products:
        raise HTTPException(404, "Product not found")

    products[product_id] = product.model_dump()

    return {"message": "Product updated"}


@app.delete("/products/{product_id}")
def delete_product(product_id: int):

    if product_id not in products:
        raise HTTPException(404, "Product not found")

    products.pop(product_id)

    return {"message": "Product deleted"}
```

**Explanation:**
- POST creates.
- GET reads.
- PUT updates.
- DELETE removes.
- `HTTPException` handles missing records and duplicate IDs.

## 40. JWT Authentication

Install:

```bash
pip install "PyJWT[crypto]"
```

Set environment variable:

```powershell
$env:JWT_SECRET_KEY="your-long-random-secret"
```

Code:

```python
import os
import jwt

from datetime import datetime, timedelta, timezone
from fastapi import FastAPI, HTTPException

app = FastAPI()

SECRET_KEY = os.environ["JWT_SECRET_KEY"]

ALGORITHM = "HS256"


def create_access_token(username: str):

    payload = {
        "sub": username,
        "exp": datetime.now(timezone.utc) + timedelta(minutes=30)
    }

    token = jwt.encode(
        payload,
        SECRET_KEY,
        algorithm=ALGORITHM
    )

    return token


def verify_token(token: str):

    try:
        payload = jwt.decode(
            token,
            SECRET_KEY,
            algorithms=[ALGORITHM]
        )

        return payload["sub"]

    except jwt.InvalidTokenError:

        raise HTTPException(
            status_code=401,
            detail="Invalid or expired token"
        )


@app.post("/login")
def login(username: str):

    token = create_access_token(username)

    return {
        "access_token": token,
        "token_type": "bearer"
    }


@app.get("/protected")
def protected_api(token: str):

    username = verify_token(token)

    return {
        "message": "Access granted",
        "username": username
    }
```

**Explanation:**
- JWT login token generate karta hai.
- `sub` username represent karta hai.
- `exp` expiry define karta hai.
- `jwt.encode()` token generate karta hai.
- `jwt.decode()` token validate karta hai.

Note: Ye learning example hai. Production mein password verification, OAuth2 Bearer authentication aur secure token handling implement karo.

## 42. SQL Database Integration

Code:

```python
from fastapi import FastAPI, Depends
from sqlalchemy import create_engine, String
from sqlalchemy.orm import (
    DeclarativeBase,
    sessionmaker,
    Session,
    Mapped,
    mapped_column
)

app = FastAPI()

DATABASE_URL = "sqlite:///./employees.db"

engine = create_engine(
    DATABASE_URL,
    connect_args={"check_same_thread": False}
)

SessionLocal = sessionmaker(
    bind=engine,
    autoflush=False,
    autocommit=False
)


class Base(DeclarativeBase):
    pass


class Employee(Base):

    __tablename__ = "employees"

    id: Mapped[int] = mapped_column(primary_key=True)

    name: Mapped[str] = mapped_column(String(100))


Base.metadata.create_all(bind=engine)


def get_db():

    db = SessionLocal()

    try:
        yield db

    finally:
        db.close()


@app.post("/employees")
def create_employee(
    name: str,
    db: Session = Depends(get_db)
):

    employee = Employee(name=name)

    db.add(employee)

    db.commit()

    db.refresh(employee)

    return {
        "id": employee.id,
        "name": employee.name
    }


@app.get("/employees")
def get_employees(db: Session = Depends(get_db)):

    return db.query(Employee).all()
```

**Explanation:**
- SQLAlchemy ORM database operations manage karta hai.
- `create_engine()` database engine create karta hai.
- `Employee` ORM model hai.
- `get_db()` database session provide karta hai.
- `db.add()` record add karta hai.
- `db.commit()` changes save karta hai.
- `db.refresh()` updated record fetch karta hai.

**Use:** PostgreSQL, MySQL, SQLite integration.

## 46. Pagination

```python
from fastapi import FastAPI, Query

app = FastAPI()

users = [
    {"id": i, "name": f"User {i}"}
    for i in range(1, 101)
]


@app.get("/users")
def get_users(
    page: int = Query(1, ge=1),
    limit: int = Query(10, ge=1, le=50)
):

    start = (page - 1) * limit

    end = start + limit

    result = users[start:end]

    return {
        "page": page,
        "limit": limit,
        "total": len(users),
        "data": result
    }
```

**Explanation:**
- `page` current page number hai.
- `limit` records per page hai.
- `start` starting index calculate karta hai.
- List slicing records return karti hai.

**Use:** Large database APIs.
