###############################################################################
# 6. DOCKERFILE INSTRUCTIONS
###############################################################################

# FROM specifies base image
# FROM python:3.12-slim

# WORKDIR sets working directory
# WORKDIR /app

# COPY copies local files into image
# COPY . .

# ADD copies files and can extract local archives
# ADD archive.tar.gz /app/

# RUN executes commands during image building
# RUN pip install fastapi uvicorn

# ENV sets environment variables
# ENV APP_ENV=production

# ARG defines build-time variables
# ARG APP_VERSION=1.0

# EXPOSE documents the application port
# EXPOSE 8000

# CMD defines the default startup command
# CMD ["python", "main.py"]

# ENTRYPOINT defines the main executable
# ENTRYPOINT ["python"]

# USER specifies the execution user
# USER 1000

# VOLUME declares a persistent data mount point
# VOLUME ["/app/data"]

# HEALTHCHECK checks application health
# HEALTHCHECK CMD curl -f http://localhost:8000/health || exit 1

# LABEL adds image metadata
# LABEL maintainer="ai-engineering-team"

# SHELL changes shell used by RUN instructions
# SHELL ["/bin/bash", "-c"]
