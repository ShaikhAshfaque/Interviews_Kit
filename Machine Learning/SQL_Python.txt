SELECT
    customer_id,
    age,
    monthly_revenue,
    tenure,
    support_calls,
    churn

AND tenure >= 12

    query,
    conn
)

# Data understanding

print(df.head())
print(df.shape)
print(df.info())
print(df.describe())

# Missing values

print(df.isnull().sum())

        "monthly_revenue",
        "tenure",
        "support_calls"
    ]
]

# Target

y = df["churn"]

print("Features:")
print(X.head())
