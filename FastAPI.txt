## 33. API Middleware

```python
from fastapi import FastAPI, Request
import time

app = FastAPI()

@app.middleware("http")
async def timing_middleware(request: Request, call_next):
    start = time.time()

    response = await call_next(request)

    response.headers["X-Process-Time"] = str(time.time() - start)

    return response

@app.get("/")
def home():
    return {"message": "Hello"}
```

**Explanation:** Middleware request aur response ke beech execute hota hai. Yahan processing time header add kiya gaya hai.

## 43. JWT Token Creation

```python
from fastapi import FastAPI
from jose import jwt
from datetime import datetime, timedelta, timezone

app = FastAPI()

SECRET_KEY = "change-this-secret"
ALGORITHM = "HS256"

@app.post("/token")
def create_token(username: str):
    payload = {
        "sub": username,
        "exp": datetime.now(timezone.utc) + timedelta(minutes=30)
    }

    token = jwt.encode(payload, SECRET_KEY, algorithm=ALGORITHM)

    return {"access_token": token}
```

**Explanation:** JWT token generate karta hai. Install `pip install "python-jose[cryptography]"`. Production mein secret environment variable mein rakhein.

## 44. JWT Token Verification

```python
from fastapi import FastAPI, HTTPException
from jose import jwt, JWTError

app = FastAPI()

SECRET_KEY = "change-this-secret"
ALGORITHM = "HS256"

@app.get("/verify-token")
def verify_token(token: str):
    try:
        payload = jwt.decode(
            token,
            SECRET_KEY,
            algorithms=[ALGORITHM]
        )

        return {"username": payload.get("sub")}

    except JWTError:
        raise HTTPException(
            status_code=401,
            detail="Invalid token"
        )
```

**Explanation:** JWT decode karke token valid hai ya nahi check karta hai.

## 47. Fetch Data from SQLite

```python
from fastapi import FastAPI
import sqlite3

app = FastAPI()

@app.get("/fetch-users")
def fetch_users():
    connection = sqlite3.connect("test.db")
    connection.row_factory = sqlite3.Row

    cursor = connection.cursor()

    cursor.execute("SELECT * FROM users")

    users = [dict(row) for row in cursor.fetchall()]

    connection.close()

    return {"users": users}
```

**Explanation:** Database se all users fetch karke JSON response return karta hai.

## 50. Complete CRUD API

```python
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel

app = FastAPI()

database = {}

class Employee(BaseModel):
    name: str
    role: str
    salary: float

@app.post("/employees/{employee_id}")
def create_employee(employee_id: int, employee: Employee):
    if employee_id in database:
        raise HTTPException(status_code=409, detail="Employee already exists")

    database[employee_id] = employee.model_dump()

    return {
        "message": "Employee created",
        "employee": database[employee_id]
    }

@app.get("/employees")
def get_employees():
    return {"employees": database}

@app.get("/employees/{employee_id}")
def get_employee(employee_id: int):
    if employee_id not in database:
        raise HTTPException(status_code=404, detail="Employee not found")

    return database[employee_id]

@app.put("/employees/{employee_id}")
def update_employee(employee_id: int, employee: Employee):
    if employee_id not in database:
        raise HTTPException(status_code=404, detail="Employee not found")

    database[employee_id] = employee.model_dump()

    return {"message": "Employee updated"}

@app.delete("/employees/{employee_id}")
def delete_employee(employee_id: int):
    if employee_id not in database:
        raise HTTPException(status_code=404, detail="Employee not found")

    del database[employee_id]

    return {"message": "Employee deleted"}
```

**Explanation:** Is program mein Create, Read, Update aur Delete operations implement kiye gaye hain. Ye in-memory demo hai; production mein proper database use karein.
