### 1. Clean text before embedding

def clean_text(text: str) -> str:

### 2. Chunk text with overlap
**Scenario:** Split long documents for a RAG pipeline.
```python
def chunk_text(text: str, size: int = 500, overlap: int = 50) -> list[str]:
    if overlap >= size:
        raise ValueError("overlap must be < size")
    chunks, step = [], size - overlap
    for i in range(0, len(text), step):
        chunks.append(text[i:i + size])
        if i + size >= len(text):
            break
    return chunks
```

### 4. Cosine similarity
**Scenario:** Compare two embedding vectors without numpy.
```python
import math

def cosine(a: list[float], b: list[float]) -> float:
    dot = sum(x * y for x, y in zip(a, b))
    na = math.sqrt(sum(x * x for x in a))
    nb = math.sqrt(sum(y * y for y in b))
    return dot / (na * nb) if na and nb else 0.0
```

### 6. Normalize vectors (L2)
**Scenario:** Vector DB expects unit-length vectors.
```python
import numpy as np

def l2_normalize(v: list[float]) -> list[float]:
    arr = np.array(v, dtype=float)
    norm = np.linalg.norm(arr)
    return (arr / norm).tolist() if norm else arr.tolist()
```

### 7. Top-k similar documents
**Scenario:** Brute-force retrieval for a small corpus.
```python
import numpy as np

def top_k(query: np.ndarray, matrix: np.ndarray, k: int = 3) -> list[tuple[int, float]]:
    sims = matrix @ query / (np.linalg.norm(matrix, axis=1) * np.linalg.norm(query))
    idx = np.argsort(-sims)[:k]
    return [(int(i), float(sims[i])) for i in idx]
```

### 8. Deduplicate documents by hash
**Scenario:** Same PDF was ingested twice.
```python
import hashlib

def dedupe(docs: list[str]) -> list[str]:
    seen, out = set(), []
    for d in docs:
        h = hashlib.sha256(d.strip().lower().encode()).hexdigest()
        if h not in seen:
            seen.add(h)
            out.append(d)
    return out
```

### 9. Flatten nested metadata
**Scenario:** Vector DB only allows flat metadata.
```python
def flatten(d: dict, parent: str = "", sep: str = ".") -> dict:
    out = {}
    for k, v in d.items():
        key = f"{parent}{sep}{k}" if parent else k
        if isinstance(v, dict):
            out.update(flatten(v, key, sep))
        else:
            out[key] = v
    return out

print(flatten({"doc": {"author": "A", "page": 2}}))  # {'doc.author': 'A', 'doc.page': 2}
```

### 11. Retry decorator
**Scenario:** LLM API fails randomly with 5xx.
```python
import time, functools

def retry(times: int = 3, delay: float = 1.0):
    def deco(fn):
        @functools.wraps(fn)
        def wrapper(*a, **kw):
            for attempt in range(1, times + 1):
                try:
                    return fn(*a, **kw)
                except Exception:
                    if attempt == times:
                        raise
                    time.sleep(delay * attempt)
        return wrapper
    return deco
```

### 12. Timing decorator
**Scenario:** Measure latency of retrieval vs generation.
```python
import time, functools

def timed(fn):
    @functools.wraps(fn)
    def wrapper(*a, **kw):
        t = time.perf_counter()
        try:
            return fn(*a, **kw)
        finally:
            print(f"{fn.__name__} took {(time.perf_counter()-t)*1000:.1f} ms")
    return wrapper
```

### 17. Sliding window for chat history
**Scenario:** Keep chat context under a token budget.
```python
def trim_history(messages: list[dict], max_tokens: int = 2000) -> list[dict]:
    total, kept = 0, []
    for m in reversed(messages):
        t = len(m["content"]) // 4
        if total + t > max_tokens:
            break
        kept.append(m)
        total += t
    return list(reversed(kept))
```

### 22. Simple rate limiter
**Scenario:** Max 5 calls per second to an LLM API.
```python
import time

class RateLimiter:
    def __init__(self, per_sec: int):
        self.interval = 1 / per_sec
        self.last = 0.0
    def wait(self):
        now = time.monotonic()
        sleep = self.last + self.interval - now
        if sleep > 0:
            time.sleep(sleep)
        self.last = time.monotonic()
```

### 25. Reciprocal Rank Fusion
**Scenario:** Combine keyword and vector search rankings.
```python
def rrf(rankings: list[list[str]], k: int = 60) -> list[str]:
    scores: dict[str, float] = {}
    for ranking in rankings:
        for rank, doc in enumerate(ranking, 1):
            scores[doc] = scores.get(doc, 0) + 1 / (k + rank)
    return sorted(scores, key=scores.get, reverse=True)

print(rrf([["a", "b", "c"], ["c", "a", "d"]]))
```

### 29. Exponential backoff on rate limit

import time, random

class RateLimitError(Exception): ...

def call_with_backoff(fn, retries=5):

        try:
            return fn()
        except RateLimitError:

### 41. Simple circuit breaker
**Scenario:** Stop calling a failing LLM provider for 30 seconds.
```python
import time

class CircuitBreaker:
    def __init__(self, threshold=3, cooldown=30):
        self.fails, self.threshold, self.cooldown, self.opened = 0, threshold, cooldown, 0.0
    def call(self, fn):
        if self.fails >= self.threshold and time.time() - self.opened < self.cooldown:
            raise RuntimeError("Circuit open")
        try:
            r = fn()
            self.fails = 0
            return r
        except Exception:
            self.fails += 1
            self.opened = time.time()
            raise
```

### 96. Rate limit per client (simple)
**Scenario:** Max 5 requests per minute per IP.
```python
import time
from collections import defaultdict, deque
from fastapi import Request, HTTPException

hits = defaultdict(deque)

def limiter(request: Request):
    ip, now = request.client.host, time.time()
    q = hits[ip]
    while q and now - q[0] > 60:
        q.popleft()
    if len(q) >= 5:
        raise HTTPException(429, "Too many requests")
    q.append(now)

@app.get("/limited", dependencies=[Depends(limiter)])

```

### 146. Cache RAG answers in Redis

import redis, json, hashlib

r = redis.Redis.from_url("redis://localhost:6379/0")

def cached_rag(question: str, fn, ttl: int = 3600):
    key = "rag:" + hashlib.sha256(question.lower().strip().encode()).hexdigest()
    if (hit := r.get(key)):
        return json.loads(hit)
    result = fn(question)
    r.setex(key, ttl, json.dumps(result))
    return result

### 147. Retry and timeout around LLM calls
**Scenario:** Make the RAG API resilient.
```python
import time
import anthropic

def safe_generate(prompt: str, retries: int = 3) -> str:
    for i in range(retries):
        try:
            r = claude.messages.create(model="claude-sonnet-5-5", max_tokens=500,
                                       messages=[{"role": "user", "content": prompt}], timeout=30)
            return r.content[0].text
        except (anthropic.RateLimitError, anthropic.APIConnectionError, anthropic.APITimeoutError):
            if i == retries - 1:
                raise
            time.sleep(2 ** i)
```

### 148. Simple retrieval evaluation (hit rate and MRR)
**Scenario:** Measure whether retrieval finds the right doc.
```python
def evaluate(test_set: list[dict], retrieve_fn, k=4):
    hits, rr = 0, 0.0
    for t in test_set:  # {"question": str, "expected_source": str}
        results = [h["source"] for h in retrieve_fn(t["question"], k)]
        if t["expected_source"] in results:
            hits += 1
            rr += 1 / (results.index(t["expected_source"]) + 1)
    n = len(test_set)
    return {"hit_rate": hits / n, "mrr": rr / n}
```

### 150. Full RAG service with fallback and logging
**Scenario:** Production-style endpoint tying it all together.
```python
import logging
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel, Field

log = logging.getLogger("rag")
app = FastAPI()

class Ask(BaseModel):
    question: str = Field(min_length=3, max_length=1000)
    top_k: int = Field(4, ge=1, le=10)

@app.post("/v1/ask")
def ask(req: Ask):
    try:
        qvec = embed_local([req.question])[0]
        hits = filter_hits(search(qvec, req.top_k * 3))
    except Exception:
        log.exception("retrieval failed")
        raise HTTPException(503, "Retrieval unavailable")

    if not hits:
        return {"answer": "I don't know based on the available documents.", "sources": []}

    docs = local_rerank(req.question, [h["text"] for h in hits], req.top_k)
    try:
        ans = safe_generate(
            "Answer only from context and cite as [n].\n\nContext:\n"
            + "\n".join(f"[{i+1}] {d}" for i, d in enumerate(docs))
            + f"\n\nQuestion: {req.question}"
        )
    except Exception:
        log.exception("generation failed")
        raise HTTPException(502, "LLM unavailable")

    log.info("question=%r hits=%d", req.question, len(docs))
    return {"answer": ans, "sources": sorted({h["source"] for h in hits})}
```
