# 1. CLASS

class Student:

    age = 21

s1 = Student()

# Object class ka instance hota hai.

class Student:
    pass

# 3. CONSTRUCTOR (__init__)
# Object create hote hi constructor automatically call hota hai.

class Student:

        self.age = age

# Self current object ko refer karta hai.

class Student:
    def __init__(self, name):
        self.name = name

    def show(self):
        print("Name:", self.name)

s1 = Student("Rahul")

# 5. INSTANCE VARIABLES
# Har object ke instance variables alag hote hain.

class Employee:

        self.salary = salary

e1 = Employee("Rahul", 30000)

# 7. INSTANCE METHODS
# Instance method self ke through object data access karta hai.

class Student:

# 8. CLASS METHODS
# Class method cls ke through class variables modify karta hai.

class Employee:

    @classmethod
    def change_company(cls, name):

Employee.change_company("Microsoft")
print("8. Class Method:", Employee.company)

# 11. ENCAPSULATION
# Data aur methods ko ek class ke andar bind karna.

class BankAccount:
    def __init__(self, balance):
        self.balance = balance

    def deposit(self, amount):
        self.balance += amount

    def show_balance(self):
        print(self.balance)

account = BankAccount(1000)
account.deposit(500)

print("11. Encapsulation:")
account.show_balance()


# 12. PUBLIC MEMBERS

# 13. PROTECTED MEMBERS

# 14. PRIVATE MEMBERS

class Bank:

# 15. GETTER AND SETTER
# Getter value read karta hai aur setter value update karta hai.

class Student:
    def __init__(self):
        self.__age = 18

    def get_age(self):
        return self.__age

    def set_age(self, age):
        if age >= 18:
            self.__age = age
        else:
            print("Invalid Age")

s1 = Student()
s1.set_age(25)

print("15. Getter Setter:", s1.get_age())


# 16. INHERITANCE
# Child class parent class ke methods inherit karti hai.

class Animal:
    def eat(self):
        print("Animal is eating")

class Dog(Animal):
    pass

d1 = Dog()

# 17. SINGLE INHERITANCE
# Ek child class ek parent class se inherit karti hai.

class Father:
    def house(self):
        print("Father owns a house")

class Son(Father):
    def car(self):
        print("Son owns a car")

s = Son()

print("17. Single Inheritance:")
s.house()
s.car()


# 18. MULTIPLE INHERITANCE
# Ek child class multiple parent classes se inherit karti hai.

class Father:
    def skills(self):
        print("Gardening")

class Mother:
    def talent(self):
        print("Cooking")

class Child(Father, Mother):
    pass

c = Child()

print("18. Multiple Inheritance:")
c.skills()
c.talent()


# 19. MULTILEVEL INHERITANCE
# Grandparent -> Parent -> Child inheritance chain.

class Grandfather:
    def show1(self):
        print("Grandfather")

class Father(Grandfather):
    def show2(self):
        print("Father")

class Son(Father):
    def show3(self):
        print("Son")

s = Son()

print("19. Multilevel Inheritance:")
s.show1()
s.show2()
s.show3()


# 20. HIERARCHICAL INHERITANCE

# 21. HYBRID INHERITANCE
# Multiple inheritance types ka combination.

class A:
    def show_a(self):
        print("Class A")

class B(A):
    def show_b(self):
        print("Class B")

class C(A):
    def show_c(self):
        print("Class C")

class D(B, C):
    def show_d(self):
        print("Class D")

obj = D()

print("21. Hybrid Inheritance:")
obj.show_a()
obj.show_b()
obj.show_c()
obj.show_d()


# 22. METHOD OVERRIDING
# Child class parent ke method ko redefine karti hai.

class Animal:
    def sound(self):
        print("Animal makes sound")

class Dog(Animal):
    def sound(self):
        print("Dog barks")

d = Dog()

print("22. Method Overriding:")
d.sound()


# 23. METHOD OVERLOADING
# Python traditional method overloading support nahi karta.
# Default arguments se similar behaviour achieve kar sakte hain.

class Calculator:
    def add(self, a, b=0, c=0):
        return a + b + c

obj = Calculator()

print("23. Method Overloading:")
print(obj.add(10, 20))
print(obj.add(10, 20, 30))


# 24. POLYMORPHISM
# Same method name, different class behaviour.

class Dog:
    def sound(self):
        print("Bark")

class Cat:
    def sound(self):
        print("Meow")

class Cow:
    def sound(self):
        print("Moo")

animals = [Dog(), Cat(), Cow()]

print("24. Polymorphism:")

for animal in animals:
    animal.sound()


# 25. DUCK TYPING
# Object ke type se zyada uske behaviour ko importance milti hai.

class Dog:
    def speak(self):
        print("Dog says Woof")

class Person:
    def speak(self):
        print("Person says Hello")

def make_sound(obj):
    obj.speak()

print("25. Duck Typing:")
make_sound(Dog())
make_sound(Person())


# 26. SUPER KEYWORD
# super() parent class ka method access karta hai.

    def show(self):
        print("Parent Method")

    def show(self):
        super().show()
        print("Child Method")

obj = Child()

print("26. Super Keyword:")
obj.show()


# 27. CONSTRUCTOR OVERRIDING

# 31. ABSTRACT CLASS

# 41. MAGIC METHODS
# Special methods Python operations ka behaviour customize karte hain.

class ShoppingCart:
    def __init__(self, items):
        self.items = items

    def __len__(self):
        return len(self.items)

    def __contains__(self, item):
        return item in self.items

cart = ShoppingCart(["Laptop", "Mobile", "Mouse"])

print("41. Magic Methods:")
print(len(cart))
print("Laptop" in cart)

# 44. __eq__ METHOD
# == operator ka custom behaviour define karta hai.

class Student:
    def __init__(self, roll):
        self.roll = roll

    def __eq__(self, other):
        if not isinstance(other, Student):
            return NotImplemented

        return self.roll == other.roll

s1 = Student(101)
s2 = Student(101)

print("44. __eq__:", s1 == s2)


# 45. CLASS INVARIANT
# Object ki valid state maintain karne ke liye conditions lagana.

class BankAccount:
    def __init__(self, balance):
        if balance < 0:
            raise ValueError("Balance cannot be negative")

        self.balance = balance

    def withdraw(self, amount):
        if amount > self.balance:
            raise ValueError("Insufficient Balance")

        self.balance -= amount

account = BankAccount(5000)
account.withdraw(1000)

print("45. Class Invariant:", account.balance)


# 46. MULTIPLE DISPATCH
# singledispatchmethod first non-self argument ke type ke basis par
# method implementation choose karta hai.

from functools import singledispatchmethod

class Printer:

    @singledispatchmethod
    def display(self, value):
        print("Other:", value)

    @display.register
    def _(self, value: int):
        print("Integer:", value)

    @display.register
    def _(self, value: str):
        print("String:", value)

p = Printer()

print("46. Multiple Dispatch:")
p.display(100)
p.display("Python")
p.display(3.14)


# 47. METACLASS
# Metaclass classes create karne wali class hoti hai.

class MyMeta(type):
    def __new__(mcs, name, bases, attrs):
        attrs["category"] = "Python OOPS"

        return super().__new__(
            mcs, name, bases, attrs
        )

class Student(metaclass=MyMeta):
    pass

print("47. Metaclass:")
print(Student.category)
print(type(Student))


# 48. DESCRIPTOR PROTOCOL
# Descriptor attribute access aur assignment control karta hai.

class PositiveNumber:

    def __set_name__(self, owner, name):
        self.name = name

    def __get__(self, obj, owner):
        if obj is None:
            return self

        return obj.__dict__[self.name]

    def __set__(self, obj, value):
        if value <= 0:
            raise ValueError("Must be positive")

        obj.__dict__[self.name] = value

class Product:
    price = PositiveNumber()

    def __init__(self, price):
        self.price = price

p = Product(500)

print("48. Descriptor:", p.price)

# 50. DEPENDENCY INJECTION
# Dependency ko class ke andar create karne ke bajaye
# externally provide kiya jata hai.

class EmailService:
    def send(self, message):
        print("Email Sent:", message)

class Notification:
    def __init__(self, service):
        self.service = service

    def notify(self, message):
        self.service.send(message)

email = EmailService()
notification = Notification(email)

print("50. Dependency Injection:")
notification.notify("Welcome to Python!")
