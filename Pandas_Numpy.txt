import pandas as pd
import numpy as np
import seaborn as sns
import matplotlib.pyplot as plt


# ==========================================
# STEP 1 — LOAD DATA
# ==========================================

df = pd.read_csv(
    "telecom_customers.csv"
)


# ==========================================
# STEP 2 — BASIC DATA UNDERSTANDING
# ==========================================

print(df.shape)

print(df.head())

print(df.info())

print(df.describe())


# ==========================================
# STEP 3 — MISSING VALUES
# ==========================================

print(
    df.isnull().sum()
)


# ==========================================
# STEP 4 — DUPLICATES
# ==========================================

print(
    df.duplicated().sum()
)

df = df.drop_duplicates()


# ==========================================
# STEP 5 — NUMERIC MISSING VALUE
# ==========================================

df["monthly_revenue"] = (
    df["monthly_revenue"]
    .fillna(
        df["monthly_revenue"].median()
    )
)


# ==========================================
# STEP 6 — CATEGORICAL MISSING VALUE
# ==========================================

df["contract_type"] = (
    df["contract_type"]
    .fillna(
        df["contract_type"].mode()[0]
    )
)


# ==========================================
# STEP 7 — CHURN DISTRIBUTION
# ==========================================

print(
    df["churn"].value_counts()
)


# ==========================================
# STEP 8 — CHURN RATE
# ==========================================

churn_rate = (
    df["churn"].mean() * 100
)

print(
    "Churn Rate:",
    churn_rate
)


# ==========================================
# STEP 9 — NUMPY STATISTICS
# ==========================================

revenue = df[
    "monthly_revenue"
].to_numpy()

print(
    "Mean Revenue:",
    np.mean(revenue)
)

print(
    "Median Revenue:",
    np.median(revenue)
)

print(
    "Std Revenue:",
    np.std(revenue)
)

print(
    "P95 Revenue:",
    np.percentile(
        revenue,
        95
    )
)


# ==========================================
# STEP 10 — OUTLIER DETECTION
# ==========================================

Q1 = np.percentile(
    revenue,
    25
)

Q3 = np.percentile(
    revenue,
    75
)

IQR = Q3 - Q1

lower = Q1 - 1.5 * IQR

upper = Q3 + 1.5 * IQR

outliers = df[
    (df["monthly_revenue"] < lower) |
    (df["monthly_revenue"] > upper)
]

print(
    "Outliers:",
    len(outliers)
)


# ==========================================
# STEP 11 — CHURN BY CONTRACT
# ==========================================

contract_churn = (
    df.groupby(
        "contract_type"
    )["churn"]
    .mean()
    .reset_index()
)

print(contract_churn)


# ==========================================
# STEP 12 — VISUALIZE CONTRACT CHURN
# ==========================================

sns.barplot(
    data=contract_churn,
    x="contract_type",
    y="churn"
)

plt.xticks(
    rotation=45
)

plt.show()


# ==========================================
# STEP 13 — REVENUE DISTRIBUTION
# ==========================================

sns.histplot(
    data=df,
    x="monthly_revenue",
    kde=True
)

plt.show()


# ==========================================
# STEP 14 — TENURE VS CHURN

)

plt.show()
